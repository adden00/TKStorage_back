const minVersionCode = 2


function doPost(e) {

  const type = e.parameter.type
  if (e.parameter.versionCode == null) {
    return ContentService.createTextOutput().append(JSON.stringify({ 'success': false, 'message': "Пожалуйста обновите приложение!" })).setMimeType(ContentService.MimeType.Text) //making error in appp
  }
  const versionCode = parseInt(e.parameter.versionCode)
  if (versionCode < minVersionCode) {
    return ContentService.createTextOutput().append(JSON.stringify({ 'success': false, 'message': "Пожалуйста обновите приложение!" })).setMimeType(ContentService.MimeType.JSON) // ask to update
  }
  if (type == 'get') {
    return getItem(e)
  }

  if (type == 'update') {
    return updateItem(e)
  }

  if (type == 'add') {
    return addItem(e)
  }

  if (type == 'getFreeId') {
    return getFreeId(e)
  }

  if (type == 'search') {
    return search(e)
  }

}

function getItem(e) {

function getFreeId() {
  const sheet = SpreadsheetApp.getActive().getSheetByName('Снаряжение 2025 актуальное')
  var lr = sheet.getLastRow()
  const lc = sheet.getLastColumn()
  const range = sheet.getRange(2, 1, lr, lc)
  range.sort(1)
  while (sheet.getRange(lr, 1).getValue() == "" && lr > 0) {
    lr--
  }
  var maxId = sheet.getRange(lr, 1).getValue()
  var freeId = maxId + 1
  var existNumber = getRowNumber(freeId, sheet)
  if (existNumber == -1 || existNumber == null) {
    return ContentService.createTextOutput().append(JSON.stringify({ 'success': true, 'id': freeId.toString() })).setMimeType(ContentService.MimeType.JSON)
  }

  // если не отсортирована, и попали в существующей, то ищем просто циклом
  lr = sheet.getLastRow()
  maxId = sheet.getRange(lr, 1).getValue()
  for (var i = 2; i <= lr; i++) {
    var rid = sheet.getRange(i, 1).getValue()
    if (rid == '') {
      break
    }
    if (Number(rid) > maxId) {
      maxId = rid
    }
  }
  if (maxId != null && maxId != "")
    return ContentService.createTextOutput().append(JSON.stringify({ 'success': true, 'id': (maxId + 1).toString() })).setMimeType(ContentService.MimeType.JSON)
  else {
    return ContentService.createTextOutput().append(JSON.stringify({ 'success': false })).setMimeType(ContentService.MimeType.JSON)
  }
}

function search(e) {
  const sheet = SpreadsheetApp.getActive().getSheetByName('Снаряжение 2025 актуальное')
  const query = e.parameter.query
  const rows = find(query, sheet)
  const items = rows.map(i => {
    return {
      'id': sheet.getRange(i, 1).getValue().toString(),
      'category': sheet.getRange(i, 2).getValue().toString(),
      'brand': sheet.getRange(i, 3).getValue().toString(),
      'name': sheet.getRange(i, 4).getValue().toString(),
      'color': sheet.getRange(i, 5).getValue().toString(),
      'weigh': sheet.getRange(i, 7).getValue().toString(),
      'quality': sheet.getRange(i, 8).getValue().toString(),
      'location': sheet.getRange(i, 9).getValue().toString(),
      'event': sheet.getRange(i, 10).getValue().toString(),
      'info': sheet.getRange(i, 12).getValue().toString(),
      'date': sheet.getRange(i, 13).getValue().toString()
    }
  })

  return ContentService.createTextOutput().append(JSON.stringify(
    {
      'success': true,
      'items': items
    }
  )).setMimeType(ContentService.MimeType.JSON)

}

function find(name, sheet) {
  const lr = sheet.getLastRow()
  const range = sheet.getRange(1, 9, lr, 1)
  const textFinder = range.createTextFinder(name)
  const searchedItems = textFinder.findAll()
  return searchedItems.map(item => {
    return item.getRow()
  })
}

function getRowNumber(id, sheet) {
  const lr = sheet.getLastRow()
  const range = sheet.getRange(1, 1, lr, 1)
  const textFinder = range.createTextFinder(id).matchEntireCell(true)
  const searchedItems = textFinder.findAll()
  if (searchedItems.length > 1) {
    //todo нашли коллизии
  }
  if (searchedItems.length == 0) {
    return -1
  }
  return searchedItems[0].getRow()
}

function sumIfDifferent(s1, s2) {
  if (s1 == s2)
    return s1
  else
    return s1 + "-->\n  ->" + s2
}

function getCurrentTime() {
  return Utilities.formatDate(new Date(), "GMT+3", "dd.MM.yyyy HH:mm")
}
